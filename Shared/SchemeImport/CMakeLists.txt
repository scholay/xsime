cmake_minimum_required(VERSION 3.27)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(RimesSchemeImport LANGUAGES C CXX)
    include(CTest)
else()
    enable_language(C)
endif()
include(FetchContent)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_CONTRIB OFF CACHE BOOL "" FORCE)
set(YAML_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(INSTALL_PROJECT OFF CACHE BOOL "" FORCE)
FetchContent_Declare(rimes_yaml
    URL https://codeload.github.com/jbeder/yaml-cpp/tar.gz/f7320141120f720aecc4c32be25586e7da9eb978
    URL_HASH SHA256=2fd3bf695ccc056835a70dd6d3046312cc1004e8338b8b5c91646918a27b7ef6)
FetchContent_Declare(rimes_miniz
    URL https://codeload.github.com/richgel999/miniz/tar.gz/77d0dce8627735138c51770d1799a1ef48f2117d
    URL_HASH SHA256=baa14349e7c5efb4a493a03af7d2f29d0ef45d87d488567fe5360edb33f8b7ba)
FetchContent_Declare(rimes_sha
    URL https://codeload.github.com/okdshin/PicoSHA2/tar.gz/161cb3fc4170fa7a3eca9e582cebd27cc4d1fe29
    URL_HASH SHA256=6cf473a00c98298d3ddee0aed853e3c799791f49dbc01c996ea46cb248e85802
    SOURCE_SUBDIR header-only)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
FetchContent_MakeAvailable(rimes_yaml rimes_miniz rimes_sha)
# yaml-cpp 0.8 predates GCC 15's removal of transitive <cstdint> includes.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(yaml-cpp PRIVATE -include cstdint)
endif()
add_library(RimesSchemeImport STATIC scheme_import.cpp)
target_compile_features(RimesSchemeImport PUBLIC cxx_std_17)
target_include_directories(RimesSchemeImport PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(RimesSchemeImport SYSTEM PRIVATE ${rimes_sha_SOURCE_DIR})
target_link_libraries(RimesSchemeImport PRIVATE yaml-cpp::yaml-cpp miniz)
set_target_properties(RimesSchemeImport PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
if(NOT ANDROID AND NOT APPLE)
    get_filename_component(RIMES_MATERIAL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
    add_library(RimesTypingMaterials STATIC "${RIMES_MATERIAL_ROOT}/Shared/TypingPractice/material_library.cpp")
    target_compile_features(RimesTypingMaterials PUBLIC cxx_std_17)
    set_target_properties(RimesTypingMaterials PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
    target_include_directories(RimesTypingMaterials PUBLIC "${RIMES_MATERIAL_ROOT}/Shared/TypingPractice")
    target_include_directories(RimesTypingMaterials SYSTEM PUBLIC "${RIMES_MATERIAL_ROOT}/platforms/windows/native/third_party")
    target_include_directories(RimesTypingMaterials SYSTEM PRIVATE ${rimes_sha_SOURCE_DIR})
    target_link_libraries(RimesTypingMaterials PRIVATE miniz)
    if(WIN32)
        target_compile_definitions(RimesTypingMaterials PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
        target_link_libraries(RimesTypingMaterials PRIVATE icu)
        if(MSVC)
            target_compile_options(RimesTypingMaterials PRIVATE /W4 /utf-8 /permissive-)
        endif()
    else()
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(TypingMaterialICU REQUIRED IMPORTED_TARGET icu-i18n icu-uc)
        target_link_libraries(RimesTypingMaterials PRIVATE PkgConfig::TypingMaterialICU)
    endif()
    if(BUILD_TESTING OR RIMES_BUILD_TESTS)
        add_executable(typing-material-tests typing_material_tests.cpp)
        set_target_properties(typing-material-tests PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
        target_link_libraries(typing-material-tests PRIVATE RimesTypingMaterials miniz)
        if(MSVC)
            target_compile_options(typing-material-tests PRIVATE /utf-8)
        endif()
        add_test(NAME TypingMaterials COMMAND typing-material-tests)
    endif()
endif()
if(WIN32)
    target_compile_definitions(RimesSchemeImport PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
    target_link_libraries(RimesSchemeImport PRIVATE normaliz)
elseif(APPLE)
    target_link_libraries(RimesSchemeImport PRIVATE "-framework CoreFoundation")
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SchemeGlib REQUIRED IMPORTED_TARGET glib-2.0)
    target_link_libraries(RimesSchemeImport PRIVATE PkgConfig::SchemeGlib)
endif()
if(MSVC)
    target_compile_options(RimesSchemeImport PRIVATE /W4 /utf-8 /permissive-)
else()
    target_compile_options(RimesSchemeImport PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING OR RIMES_BUILD_TESTS)
    enable_testing()
    add_executable(scheme-import-tests scheme_import_tests.cpp)
    if(MSVC)
        # Fixture paths and text must match the UTF-8 production importer.
        # The parent library's PRIVATE compiler option does not reach tests.
        target_compile_options(scheme-import-tests PRIVATE /utf-8)
    endif()
    target_link_libraries(scheme-import-tests PRIVATE RimesSchemeImport miniz)
    add_test(NAME scheme-import COMMAND scheme-import-tests)
endif()
